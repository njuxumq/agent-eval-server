# 性能测试方案

## 一、测试目标

### 1.1 性能指标目标

| 指标类别 | 指标名称 | 目标值 | 说明 |
|----------|----------|--------|------|
| **吞吐量** | 任务提交TPS | ≥ 100 TPS | 每秒任务提交数 |
| **吞吐量** | 子任务分发TPS | ≥ 500 TPS | 每秒子任务分发数 |
| **吞吐量** | 状态上报TPS | ≥ 1000 TPS | 每秒状态上报数 |
| **延迟** | 任务提交P99延迟 | ≤ 200ms | 从请求到响应 |
| **延迟** | 任务查询P99延迟 | ≤ 100ms | 从请求到响应 |
| **延迟** | 子任务分发P99延迟 | ≤ 100ms | 从分发到确认 |
| **并发** | 最大并发任务数 | ≥ 1000 | 同时运行的任务 |
| **并发** | 单执行服务并发子任务数 | ≥ 50 | 单节点并发能力 |
| **资源** | 调度服务CPU使用率 | ≤ 70% | 目标负载下 |
| **资源** | 执行服务CPU使用率 | ≤ 70% | 目标负载下 |
| **资源** | 调度服务内存使用 | ≤ 2GB | 目标负载下 |
| **资源** | 执行服务内存使用 | ≤ 1GB | 目标负载下 |
| **可靠性** | 错误率 | ≤ 0.1% | 请求失败比例 |
| **可靠性** | 可用性 | ≥ 99.9% | 服务可用时间比例 |

### 1.2 测试环境要求

| 组件 | 规格 | 数量 |
|------|------|------|
| 调度服务 | 4核8GB | 2 |
| 执行服务 | 4核8GB | 5 |
| MongoDB | 4核8GB, SSD | 3 (副本集) |
| AgentCube | 按需 | 共享集群 |
| 压测客户端 | 8核16GB | 2 |

```javascript
// test/performance/scripts/task_submit.js
import http from 'k6/http';
import { check, sleep } from 'k6';
import { uuidv4 } from 'https://jslib.k6.io/k6-utils/1.4.0/index.js';
import { Rate, Trend } from 'k6/metrics';

// 自定义指标
const errorRate = new Rate('errors');
const taskCreateLatency = new Trend('task_create_latency');

export const options = {
  stages: [
    { duration: '2m', target: 10 },   // 低负载
    { duration: '3m', target: 50 },   // 中负载
    { duration: '3m', target: 100 },  // 高负载
    { duration: '2m', target: 200 },  // 峰值负载
    { duration: '2m', target: 0 },    // 恢复
  ],
  thresholds: {
    http_req_duration: ['p(99)<200'], // P99延迟 < 200ms
    http_req_failed: ['rate<0.001'],  // 错误率 < 0.1%
    errors: ['rate<0.001'],
  },
};

const BASE_URL = __ENV.SCHEDULER_URL || 'http://localhost:8080';

export default function () {
  const appId = `perf-test-${uuidv4()}`;
  
  const payload = JSON.stringify({
    app_id: appId,
    description: '性能测试任务',
    config: {
      version: '1.0.0',
      workflow: ['synthesis'],
      models: [
        {
          id: 'test-model',
          type: 'api-openai',
          api_key: 'test-key',
          api_url: 'http://mock-server:8080',
          model: 'test-model',
          concurrency: 10,
        },
      ],
      synthesis: {
        type: 'agent',
        count: 5,
        model: 'test-model',
      },
    },
    resources: {
      skills: '',
      files: {},
    },
  });

  const params = {
    headers: {
      'Content-Type': 'application/json',
    },
    tags: { operation: 'create_task' },
  };

  const startTime = new Date();
  const response = http.post(`${BASE_URL}/v1/tasks`, payload, params);
  const duration = new Date() - startTime;
  
  taskCreateLatency.add(duration);

  const success = check(response, {
    'status is 200': (r) => r.status === 200,
    'has task_id': (r) => {
      try {
        const body = r.json();
        return body.data && body.data.task_id !== undefined;
      } catch (e) {
        return false;
      }
    },
    'response time < 200ms': (r) => r.timings.duration < 200,
  });

  errorRate.add(!success);

  sleep(1);
}

// 测试后清理
export function teardown(data) {
  console.log('Performance test completed');
}
```

```javascript
// test/performance/scripts/concurrent_execution.js
import http from 'k6/http';
import { check, sleep } from 'k6';
import { SharedArray } from 'k6/data';

// 加载测试数据
const testCases = new SharedArray('test cases', function () {
  return JSON.parse(open('./test_cases.json'));
});

export const options = {
  scenarios: {
    // 场景1: 持续提交任务
    continuous_submit: {
      executor: 'constant-arrival-rate',
      rate: 10,
      timeUnit: '1s',
      duration: '30m',
      preAllocatedVUs: 50,
      maxVUs: 200,
    },
    // 场景2: 查询任务状态
    query_status: {
      executor: 'constant-arrival-rate',
      rate: 50,
      timeUnit: '1s',
      duration: '30m',
      preAllocatedVUs: 20,
      maxVUs: 100,
      exec: 'queryTaskStatus',
    },
  },
  thresholds: {
    'http_req_duration{operation:submit}': ['p(99)<200'],
    'http_req_duration{operation:query}': ['p(99)<100'],
  },
};

const BASE_URL = __ENV.SCHEDULER_URL || 'http://localhost:8080';
let taskIds = [];

export function submitTask() {
  const testCase = testCases[Math.floor(Math.random() * testCases.length)];
  
  const payload = JSON.stringify({
    app_id: `perf-test-${Date.now()}`,
    config: testCase,
    resources: { skills: '', files: {} },
  });

  const response = http.post(`${BASE_URL}/v1/tasks`, payload, {
    headers: { 'Content-Type': 'application/json' },
    tags: { operation: 'submit' },
  });

  if (response.status === 200) {
    try {
      const body = response.json();
      if (body.data && body.data.task_id) {
        taskIds.push(body.data.task_id);
      }
    } catch (e) {}
  }

  sleep(1);
}

export function queryTaskStatus() {
  if (taskIds.length === 0) {
    sleep(0.1);
    return;
  }

  const taskId = taskIds[Math.floor(Math.random() * taskIds.length)];
  
  const response = http.get(`${BASE_URL}/v1/tasks/${taskId}?app_id=perf-test`, {
    tags: { operation: 'query' },
  });

  check(response, {
    'query status 200': (r) => r.status === 200,
  });

  sleep(0.1);
}

export default function () {
  submitTask();
}
```

```go
    m.mu.Lock()
    defer m.mu.Unlock()
    
    if s, ok := m.sandboxes[sb.SessionID]; ok {
        s.Files[remotePath] = content
    }
    
    return nil
}

    m.mu.Lock()
    defer m.mu.Unlock()
```

```yaml
# .github/workflows/performance-test.yml
name: Performance Test

on:

        run: |
          curl https://github.com/grafana/k6/releases/download/v0.45.0/k6-v0.45.0-linux-amd64.tar.gz -L | tar xvz
          sudo mv k6-*/k6 /usr/local/bin/
      
      - name: Start test environment
```

## 八、最佳实践

### 8.1 测试数据管理

- 使用独立的测试数据库，避免污染生产数据
- 每次测试前清理历史数据
- 使用固定的测试数据集，保证测试可重复

### 8.2 测试执行策略

- 先执行单接口压测，再执行混合压测
- 逐步增加负载，观察系统响应
- 记录每次测试的配置和结果

### 8.3 结果分析

- 关注P99延迟而非平均值
- 分析错误分布和错误类型
- 对比不同版本的测试结果

### 8.4 持续优化

- 定期执行性能回归测试
- 监控性能指标趋势
- 建立性能基线，及时发现退化
